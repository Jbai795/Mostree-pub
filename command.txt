    void SS3Dtree::prepareMaskedRevealBatch(
    const std::vector<uint32_t>& alphas,
    aby3::su8Matrix& ssDGKSchema,
    std::vector<u8>& my_to_peer)
    {
        if (dpfRdxShares.size() != dpfN)
            throw std::runtime_error("prepareMaskedRevealBatch: dpfRdxShares not ready");

        my_to_peer.resize(alphas.size());

        for (size_t i = 0; i < alphas.size(); ++i) {
            uint32_t alpha = alphas[i];
            if (alpha >= dpfN)
                throw std::runtime_error("prepareMaskedRevealBatch: alpha out of range");

            u8 rdx = dpfRdxShares[alpha];

            if (partyIdx() == 1) my_to_peer[i] = (u8)(ssDGKSchema(0,0)[1] + rdx);
            else                 my_to_peer[i] = (u8)(ssDGKSchema(0,0)[0] + rdx);
        }
    }

    // p0 reveals masked values to p1 and p2
    void SS3Dtree::sendMaskedRevealBatch (aby3::su8Matrix&ssDGKSchema, const std::vector<u8> rdx_to_pre, const std::vector<u8> rdx_to_next){
        if (partyIdx() != 0) return;

        uint64_t sz = rdx_to_pre.size();
        if (rdx_to_pre.size() != rdx_to_next.size())
        throw std::runtime_error("sendMaskedRevealBatch: size mismatch");
        std::vector<u8> send_to_next; // send to p1
        std::vector<u8> send_to_pre; // send to p2

        send_to_next.resize(dpfN);
        send_to_pre.resize(dpfN);
        
        for (size_t i = 0; i < dpfN; ++i) {
            send_to_next[i] = (u8)(ssDGKSchema(i,0)[0] + rdx_to_next[i]);
            send_to_pre[i] = (u8)(ssDGKSchema(i,0)[1] + rdx_to_pre[i]);
        }
        
        // to party1 
        mNext.send(sz);
        mNext.send(rdx_to_pre.data(), sz);

        // to party2
        mPrev.send(sz);
        mPrev.send(rdx_to_next.data(), sz);
    }
